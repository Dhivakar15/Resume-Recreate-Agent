    81→function timeAgo(iso: string): string {
    82→  const diff = Date.now() - new Date(iso).getTime();
    83→  const mins = Math.floor(diff / 60000);
    84→  if (mins < 1) return 'just now';
    85→  if (mins < 60) return `${mins}m ago`;
    86→  const hours = Math.floor(mins / 60);
    87→  if (hours < 24) return `${hours}h ago`;
    88→  const days = Math.floor(hours / 24);
    89→  if (days < 30) return `${days}d ago`;
    90→  return new Date(iso).toLocaleDateString();
    91→}

    95→function ScoreRing({ value, size = 84, label }: { value: number; size?: number; label?: string }) {
    96→  const stroke = 7;
    97→  const r = (size - stroke) / 2;
    98→  const c = 2 * Math.PI * r;
    99→  const pct = Math.max(0, Math.min(100, value));
   100→  return (
   101→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   102→      <svg width={size} height={size} className="-rotate-90" role="img" aria-label={label ?? `${pct} out of 100`}>
   103→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(255,255,255,0.08)" strokeWidth={stroke} />
   104→        <circle
   105→          cx={size / 2}
   106→          cy={size / 2}
   107→          r={r}
   108→          fill="none"
   109→          stroke="url(#ringGrad)"
   110→          strokeWidth={stroke}
   111→          strokeLinecap="round"
   112→          strokeDasharray={`${(pct / 100) * c} ${c}`}
   113→          className="transition-[stroke-dasharray] duration-700"
   114→        />
   115→        <defs>
   116→          <linearGradient id="ringGrad" x1="0%" y1="0%" x2="100%" y2="100%">
   117→            <stop offset="0%" stopColor="#8b5cf6" />
   118→            <stop offset="55%" stopColor="#d946ef" />
   119→            <stop offset="100%" stopColor="#22d3ee" />
   120→          </linearGradient>
   121→        </defs>
   122→      </svg>
   123→      <span className="absolute text-lg font-bold text-white">{pct}</span>
   124→    </div>
   125→  );
   126→}
